def iniciar_navegador(driver, email, senha):
    try:
        # Acessa a página de login do 99Freelas
        print("Iniciando o navegador no modo indetectável...")

        # Usamos o Driver do SeleniumBase sem o bloco 'with' para que o
        # navegador continue aberto após o fim desta funçã
        url = "https://www.99freelas.com.br/login"

        # Abre a página desconectando o driver estrategicamente para evitar detecção
        driver.uc_open_with_reconnect(url, reconnect_time=6)

        # Se houver um CAPTCHA simples (como o Turnstile da Cloudflare), ele tenta clicar sozinho
        driver.uc_gui_click_captcha()

    except Exception as e:
        print(f"Erro ao tentar lidar com CAPTCHA: {e}")

    try:
        # Insere o email e a senha
        driver.type("//*[@id='email']", email)
        driver.type("//*[@id='senha']", senha)

        # Submete o formulário pressionando ENTER
        driver.click("//*[@id='btnEfetuarLogin']")

        print("Aguardando a mudança de URL após o login...")

        # Espera inserir o código de duas etapas
        driver.sleep(20)

    except Exception as e:
        print(f"Erro ao tentar preencher o formulário de login: {e}")

    try:
        print("Login realizado com sucesso!")
        return driver

    except Exception as e:
        print(f"Erro ao aguardar a mudança de URL após o login: {e}")
        return None

def eh_projeto_exclusivo(driver):
    """
    Verifica se o projeto na posição 'num' é exclusivo com base na imagem do selo.
    Retorna True se for exclusivo, False caso contrário ou se a imagem não existir.
    """

    # Monta o XPath dinamicamente utilizando f-string com o número do índice
    xpath_imagem = f'img[alt="Projeto exclusivo"]'
    try:
        # 1. Verifica silenciosamente se a imagem existe (Retorna True ou False na hora, sem erros)
        if driver.is_element_present(xpath_imagem):

            # 2. Pega o atributo 'src' diretamente usando o XPath
            src = driver.get_attribute(xpath_imagem, "src")

            # 3. Verifica se a URL da imagem contém o nome do selo
            if src and "flat_project_exclusive.png" in src:
                return True

    except NoSuchElementException:
        # Se a imagem não for encontrada no HTML, captura a exceção
        # e permite que a função siga para retornar False
        pass

    # Retorna False para qualquer outro cenário (não tem a string ou a imagem não existe)
    return False


def obter_descricao_projeto(driver, num):
    texto_descricao = False

    xpath_titulo = f"div:nth-of-type(3) > ul > li:nth-of-type({num}) hgroup h1 a"
    seletor_descricao = "div.item-text.project-description.formatted-text"

    if not driver.is_element_present(xpath_titulo):
        return False

    url_projeto = driver.get_attribute(xpath_titulo, "href")
    if not url_projeto:
        return False

    try:
        driver.execute_script(f"window.open('{url_projeto}', '_blank');")
        driver.switch_to_newest_window()

        driver.wait_for_element_visible(seletor_descricao, timeout=10)
        texto_descricao = driver.get_text(seletor_descricao)
    except Exception:
        texto_descricao = False

    return texto_descricao


def gerar_proposta_gemini(driver, descricao_projeto):
    """
    Recebe a descrição do projeto, pede a proposta ao Gemini em formato JSON
    e tenta novamente caso os servidores do Google estejam sobrecarregados.
    """
    try:
        contratante = driver.get_text("span.name:first-child")
    except:
        contratante = "Cliente"

    prompt = f"""
Você é um desenvolvedor freelancer especializado em automação e engenharia de dados. Com base na descrição do projeto abaixo, crie uma proposta de trabalho.

Siga RIGOROSAMENTE as seguintes regras:
1. Retorne EXCLUSIVAMENTE um objeto JSON válido. NENHUM texto fora do JSON.
2. O JSON deve conter exatamente duas chaves:
- "proposta": O texto completo da proposta a ser enviada ao cliente.
- "tempo_dias": Apenas um número inteiro representando a estimativa de dias para entregar o projeto.

Regras para o texto da chave "proposta":
- Escreva de forma humana, direta e profissional.
- Faça uma estimativa do valor que cobraria pelo projeto, mas não inclua o valor no JSON. O valor deve ser mencionado apenas no texto da proposta, de forma natural e ele não pode ser um valor absurdamente alto porque eu sou um iniciante. Sugira um valor justo e competitivo.
- Evite repetir palavras ou frases.
- Nunca use asteriscos (*) ou formatações markdown.
- Inclua a estimativa de tempo no texto de forma natural.
- Informe que somos uma equipe de 3 desenvolvedores: Lukas, Yohan, Alonso e que podemos entregar o projeto com qualidade.
- Insira meus contatos: Yohan Vallier / yohanvallier01@gmail.com / WhatsApp: 21965156704

- Nome do Contratante: {contratante}
- Descrição do Projeto: {descricao_projeto}
"""

    # Lógica de Retry: Tentar até 3 vezes se o servidor estiver sobrecarregado
    tentativas = 3

    for tentativa in range(1, tentativas + 1):
        try:
            client = genai.Client(api_key=os.getenv("GOOGLE_API_KEY"))
            gemini_model = os.getenv("GEMINI_MODEL")

            configuracao_seguranca = types.GenerateContentConfig(
                response_mime_type="application/json",
                safety_settings=[
                    types.SafetySetting(
                        category=types.HarmCategory.HARM_CATEGORY_HARASSMENT,
                        threshold=types.HarmBlockThreshold.BLOCK_NONE,
                    ),
                    types.SafetySetting(
                        category=types.HarmCategory.HARM_CATEGORY_HATE_SPEECH,
                        threshold=types.HarmBlockThreshold.BLOCK_NONE,
                    ),
                    types.SafetySetting(
                        category=types.HarmCategory.HARM_CATEGORY_SEXUALLY_EXPLICIT,
                        threshold=types.HarmBlockThreshold.BLOCK_NONE,
                    ),
                    types.SafetySetting(
                        category=types.HarmCategory.HARM_CATEGORY_DANGEROUS_CONTENT,
                        threshold=types.HarmBlockThreshold.BLOCK_NONE,
                    ),
                ],
            )

            response = client.models.generate_content(
                model=gemini_model, contents=prompt, config=configuracao_seguranca
            )

            dados_retornados = json.loads(response.text)
            texto_proposta = dados_retornados.get(
                "proposta", "Não foi possível gerar a proposta."
            )
            tempo_gemini = dados_retornados.get("tempo_dias", 0)

            return texto_proposta, tempo_gemini

        except Exception as e:
            erro_str = str(e)
            # Verifica se o erro é o 503 (Servidor indisponível) ou 429 (Muitas requisições)
            if "503" in erro_str or "429" in erro_str:
                print(
                    f"⚠️ Servidor do Gemini sobrecarregado (Tentativa {tentativa}/{tentativas}). Aguardando 10 segundos..."
                )
                if tentativa < tentativas:
                    time.sleep(10)  # Espera 10 segundos antes de tentar de novo
                    continue  # Volta para o começo do For

            # Se for outro tipo de erro (como erro de validação JSON) ou se acabaram as tentativas
            print(f"❌ Erro Crítico na API do Gemini: {e}")
            return "Não foi possível gerar a proposta no momento.", 0

    return "Não foi possível gerar a proposta devido à instabilidade do servidor.", 0

def voltar_para_lista_de_projetos(driver):
    """Fecha todas as abas extras e garante o foco na primeira aba (lista de projetos)."""
    # Enquanto houver mais de 1 aba aberta, fecha a mais recente
    while len(driver.driver.window_handles) > 1:
        driver.switch_to_newest_window()
        driver.driver.close()  # Comando oficial do SeleniumBase para fechar a aba
        driver.sleep(0.5)

    # Garante que o foco retorne para a aba principal (índice 0)
    driver.switch_to_window(0)


def raspar_propostas_ja_enviadas(driver):
    """
    Clica repetidamente no botão para carregar todas as propostas anteriores
    até ele desaparecer, e depois raspa todos os títulos.
    """
    titulos_salvos = []
    xpath_botao_mais = '//*[@id="minhas-propostas"]/div[2]/div[2]'
    xpath_ul = '//*[@id="minhas-propostas"]/div[2]/ul'

    print("Verificando propostas já enviadas anteriormente...")
    driver.sleep(3)  # Aguarda o carregamento do painel principal

    try:
        # 1. Clicar no botão até ele desaparecer do HTML ou ficar invisível
        print("Expandindo o histórico de propostas...")
        tentativas = 0

        # O loop continua enquanto o botão estiver visível na tela e não passar de 50 cliques
        while driver.is_element_visible(xpath_botao_mais) and tentativas < 50:
            try:
                driver.click(xpath_botao_mais)
                print(
                    f"Botão clicado ({tentativas + 1}). Aguardando carregar mais itens..."
                )
                driver.sleep(
                    2.5
                )  # Tempo essencial para o 99Freelas processar os novos 'li' na tela
                tentativas += 1
            except Exception as e:
                print("O botão parou de responder ou desapareceu.")
                break

        print("Todo o histórico foi expandido na tela.")

        # 2. Extrair os títulos de todos os 'li' carregados dentro do 'ul'
        if driver.is_element_present(xpath_ul):
            elementos_li = driver.find_elements(f"{xpath_ul}/li")
            total_propostas = len(elementos_li)

            print(f"Foram encontradas {total_propostas} propostas na sua conta.")

            for i in range(1, total_propostas + 1):
                # O XPath desce no 'li' específico e procura a tag 'a' que guarda o título
                xpath_titulo_proposta = f"{xpath_ul}/li[{i}]//a"

                if driver.is_element_present(xpath_titulo_proposta):
                    titulo = driver.get_text(xpath_titulo_proposta).strip()
                    if titulo:
                        titulos_salvos.append(titulo)

            print(
                f"Histórico salvo na memória: {len(titulos_salvos)} títulos mapeados e guardados."
            )
        else:
            print(
                "Aviso: A lista de 'minhas-propostas' está vazia ou não foi encontrada."
            )

    except Exception as e:
        print(f"Aviso ao raspar histórico de propostas: {e}")

    return titulos_salvos


def validar_por_regex(texto, dicionario_filtros):
    """
    Verifica se o texto contém alguma das palavras-chave do dicionário usando Regex.
    """
    if not texto:
        return False

    # Junta todas as palavras-chave de todas as categorias em uma única lista
    todas_palavras = []
    for palavras in dicionario_filtros.values():
        todas_palavras.extend(palavras)

    # Cria a regra Regex:
    # (?i) -> Ignora maiúsculas e minúsculas (case-insensitive)
    # \b   -> Word boundary (garante que ache 'bot' e não 'boteco' ou 'sabotagem')
    # re.escape -> Protege contra caracteres especiais (como espaços)
    padrao = r"(?i)\b(?:" + "|".join(map(re.escape, todas_palavras)) + r")\b"

    # Se encontrar alguma palavra no texto, retorna True
    if re.search(padrao, texto):
        return True

    return False
